# 学校数据接入开发指南

> **目标**：实现与内蒙古农业大学数据库的数据同步，支持用户信息、组织关系等数据的获取  
> **接入方式**：API接口、ETL数据库直连  
> **开发周期**：3-5个工作日  

---

## 一、开发准备

### 1.1 环境要求

- **Java**：JDK 11+
- **Spring Boot**：2.7+
- **数据库**：MySQL 8.0+
- **连接池**：HikariCP
- **调度器**：Spring Task Scheduler

### 1.2 前置条件

1. **获取学校数据接入权限**：
   ```bash
   API接入地址: https://api.imau.edu.cn
   API访问令牌: your_api_token
   数据库连接信息: jdbc_url, username, password
   数据同步频率: 每日凌晨2点
   ```

2. **了解数据结构**：
   - 用户基础信息表结构
   - 组织关系表结构
   - 数据更新频率和字段含义

```java
@ConditionalOnProperty(name = "school.data.source", havingValue = "api")
public class ApiClientConfig {
    
    @Autowired

    @Bean
    public RestTemplate schoolApiRestTemplate() {
        RestTemplate restTemplate = new RestTemplate();
        
        // 配置超时
        HttpComponentsClientHttpRequestFactory factory = 
            new HttpComponentsClientHttpRequestFactory();
        factory.setConnectTimeout(properties.getApi().getTimeout());
        factory.setReadTimeout(properties.getApi().getTimeout());
        
        restTemplate.setRequestFactory(factory);
        
        // 配置拦截器
        restTemplate.setInterceptors(Arrays.asList(
            new ApiAuthInterceptor(properties.getApi().getToken()),
            new LoggingInterceptor()
        ));
        
        return restTemplate;
    }
    
    @Component
    public static class ApiAuthInterceptor implements ClientHttpRequestInterceptor {
        
        private final String token;
        
        public ApiAuthInterceptor(String token) {
            this.token = token;
        }
        
        @Override
        public ClientHttpResponse intercept(HttpRequest request, byte[] body, 
                                       ClientHttpRequestExecution execution) throws IOException {
            request.getHeaders().setBearerAuth(token);
            request.getHeaders().setContentType(MediaType.APPLICATION_JSON);
            return execution.execute(request, body);
        }
    }
}
```

```java
@Component
@ConditionalOnProperty(name = "school.data.source", havingValue = "api")
public class SchoolApiClient implements SchoolDataClient {
    
    @Autowired
    private RestTemplate restTemplate;
    
    @Autowired
    private SchoolDataProperties properties;
    
    @Override
    public List<SchoolUser> getAllUsers() {
        return fetchWithRetry("/users", SchoolUser[].class);
    }
    
    @Override
    public List<SchoolUser> getUsersByPage(int page, int size) {
        String url = String.format("/users?page=%d&size=%d", page, size);
        return fetchWithRetry(url, SchoolUser[].class);
    }
    
    @Override
    public SchoolUser getUserByUsername(String username) {
        String url = "/users/" + username;
        return fetchWithRetry(url, SchoolUser.class);
    }
    
    @Override
    public List<SchoolOrganization> getAllOrganizations() {
        return fetchWithRetry("/organizations", SchoolOrganization[].class);
    }
    
    @Override
    public List<SchoolUser> getIncrementalUsers(LocalDateTime lastSyncTime) {
        String url = "/users/incremental?since=" + 
            lastSyncTime.format(DateTimeFormatter.ISO_DATE_TIME);
        return fetchWithRetry(url, SchoolUser[].class);
    }
    
    private <T> List<T> fetchWithRetry(String endpoint, Class<T[]> responseType) {
        String url = properties.getApi().getBaseUrl() + endpoint;
        
        for (int attempt = 1; attempt <= properties.getApi().getRetryAttempts(); attempt++) {
            try {
                ResponseEntity<T[]> response = restTemplate.getForEntity(url, responseType);
                
                if (response.getStatusCode().is2xxSuccessful()) {
                    return Arrays.asList(Objects.requireNonNull(response.getBody()));
                } else {
                    log.warn("API请求失败，状态码: {}", response.getStatusCode());
                }
                
            } catch (Exception e) {
                log.error("API请求异常，尝试次数: {}/{}, URL: {}", 
                    attempt, properties.getApi().getRetryAttempts(), url, e);
                
                if (attempt == properties.getApi().getRetryAttempts()) {
                    throw new RuntimeException("API请求失败，已达到最大重试次数", e);
                }
                
                // 等待后重试
                try {
                    Thread.sleep(1000 * attempt);
                } catch (InterruptedException ie) {
                    Thread.currentThread().interrupt();
                    throw new RuntimeException("重试被中断", ie);
                }
            }
        }
        
        throw new RuntimeException("API请求失败");
    }
    
    private <T> T fetchWithRetry(String endpoint, Class<T> responseType) {
        List<T> list = fetchWithRetry(endpoint, (Class<T[]>) responseType);
        return list.isEmpty() ? null : list.get(0);
    }
}
```

```java
                org_name,
                org_type,

            org.setOrgId(rs.getString("org_id"));
            org.setOrgName(rs.getString("org_name"));
            org.setOrgType(rs.getString("org_type"));
            org.setParentId(rs.getString("parent_id"));
            org.setOrgLevel(rs.getInt("org_level"));
            org.setSortOrder(rs.getInt("sort_order"));
            org.setStatus(rs.getString("status"));
            return org;
        });
    }

    public List<SchoolUser> getIncrementalUsers(LocalDateTime lastSyncTime) {
        String sql = """
            SELECT 
                username,
                student_id,
                employee_id,
                real_name,
                email,
                phone,
                college,
                major,
                class_name,
                grade,
                user_type,
                status,
                organizations,
                raw_data

            WHERE status = 'active'

    private List<SchoolUser> queryUsers(String sql, Object... params) {
        return jdbcTemplate.query(sql, (rs, rowNum) -> {
            SchoolUser user = new SchoolUser();
            user.setUsername(rs.getString("username"));
            user.setStudentId(rs.getString("student_id"));
            user.setEmployeeId(rs.getString("employee_id"));
            user.setRealName(rs.getString("real_name"));
            user.setEmail(rs.getString("email"));

            user.setCollege(rs.getString("college"));
            user.setMajor(rs.getString("major"));
            user.setClassName(rs.getString("class_name"));
            user.setGrade(rs.getString("grade"));
            user.setUserType(rs.getString("user_type"));
            user.setStatus(rs.getString("status"));

                String organizationsJson = rs.getString("organizations");

                    ObjectMapper mapper = new ObjectMapper();
                    user.setRawData(mapper.readValue(rawDataJson, Map.class));
                }
            } catch (Exception e) {
                log.warn("解析JSON字段失败: {}", e.getMessage());
            }
            
            return user;
        }, params);
    }
```

### 5.2 数据同步实现

#### DataSyncService.java
```java
@Service
@ConditionalOnProperty(name = "school.data.sync.enabled", havingValue = "true")
public class DataSyncService {
    
    @Autowired
    private SchoolDataService schoolDataService;
    
    @Autowired
    private FieldMappingService fieldMappingService;
    
    @Autowired
    private SyncLogRepository syncLogRepository;
    
    @Autowired
    private SchoolUserSyncRepository userSyncRepository;
    
    /**
     * 执行全量同步
     */
    @Transactional
    public SyncResult performFullSync() {
        SyncLog syncLog = createSyncLog("full");
        
        try {
            // 同步用户数据
            SyncResult userResult = syncUsers(true);
            
            // 同步组织数据
            SyncResult orgResult = syncOrganizations();
            
            // 合并结果
            SyncResult totalResult = SyncResult.merge(userResult, orgResult);
            
            // 更新同步日志
            updateSyncLog(syncLog, totalResult, "success");
            
            return totalResult;
            
        } catch (Exception e) {
            log.error("全量同步失败", e);
            updateSyncLog(syncLog, null, "failed", e.getMessage());
            throw new RuntimeException("全量同步失败", e);
        }
    }
    
    /**
     * 执行增量同步
     */
    @Transactional
    public SyncResult performIncrementalSync() {
        SyncLog syncLog = createSyncLog("incremental");
        
        try {
            // 获取上次同步时间
            LocalDateTime lastSyncTime = getLastSyncTime();
            
            // 执行增量同步
            SyncResult result = syncIncrementalUsers(lastSyncTime);
            
            // 更新同步日志
            updateSyncLog(syncLog, result, "success");
            
            return result;
            
        } catch (Exception e) {
            log.error("增量同步失败", e);
            updateSyncLog(syncLog, null, "failed", e.getMessage());
            throw new RuntimeException("增量同步失败", e);
        }
    }
    
    private SyncResult syncUsers(boolean fullSync) {
        List<SchoolUser> schoolUsers;
        
        if (fullSync) {
            schoolUsers = schoolDataService.getAllUsers();
        } else {
            LocalDateTime lastSyncTime = getLastSyncTime();
            schoolUsers = schoolDataService.getIncrementalUsers(lastSyncTime);
        }
        
        SyncResult result = new SyncResult();
        result.setTotalRecords(schoolUsers.size());
        
        for (SchoolUser schoolUser : schoolUsers) {
            try {
                // 字段映射
                SchoolUserSync userSync = fieldMappingService.mapToUserSync(schoolUser);
                
                // 保存或更新
                Optional<SchoolUserSync> existing = userSyncRepository
                    .findByUsername(schoolUser.getUsername());
                
                if (existing.isPresent()) {
                    updateUserSync(existing.get(), userSync);
                    result.incrementUpdated();
                } else {
                    userSyncRepository.save(userSync);
                    result.incrementCreated();
                }
                
            } catch (Exception e) {
                log.error("同步用户失败: {}", schoolUser.getUsername(), e);
                result.incrementErrors();
            }
        }
        
        return result;
    }
    
    private SyncResult syncOrganizations() {
        List<SchoolOrganization> organizations = schoolDataService.getAllOrganizations();
        
        SyncResult result = new SyncResult();
        result.setTotalRecords(organizations.size());
        
        for (SchoolOrganization org : organizations) {
            try {
                // 保存或更新组织
                Optional<SchoolOrgSync> existing = orgSyncRepository
                    .findByOrgId(org.getOrgId());
                
                if (existing.isPresent()) {
                    updateOrgSync(existing.get(), org);
                    result.incrementUpdated();
                } else {
                    SchoolOrgSync orgSync = mapToOrgSync(org);
                    orgSyncRepository.save(orgSync);
                    result.incrementCreated();
                }
                
            } catch (Exception e) {
                log.error("同步组织失败: {}", org.getOrgId(), e);
                result.incrementErrors();
            }
        }
        
        return result;
    }
    
    private LocalDateTime getLastSyncTime() {
        Optional<SyncLog> lastSync = syncLogRepository
            .findFirstBySyncTypeOrderByStartTimeDesc("incremental");
        
        return lastSync.map(SyncLog::getStartTime)
            .orElse(LocalDateTime.now().minusDays(1));
    }
}
```

```java
@Service
public class FieldMappingService {
    
    @Autowired
    private FieldMappingRepository mappingRepository;
    
    @Autowired
    private SchoolDataProperties properties;
    
    /**
     * 将学校用户映射到本地同步表
     */
    public SchoolUserSync mapToUserSync(SchoolUser schoolUser) {
        SchoolUserSync userSync = new SchoolUserSync();
        
        // 获取字段映射配置
        List<FieldMapping> mappings = mappingRepository
            .findBySyncSourceAndIsActive(properties.getSource(), true);
        
        // 应用字段映射
        for (FieldMapping mapping : mappings) {
            try {
                Object value = getFieldValue(schoolUser, mapping.getSchoolField());
                if (value != null) {
                    Object convertedValue = convertValue(value, mapping);
                    setFieldValue(userSync, mapping.getLocalField(), convertedValue);
                } else if (mapping.getDefaultValue() != null) {
                    setFieldValue(userSync, mapping.getLocalField(), mapping.getDefaultValue());
                }
                
                if (mapping.getIsRequired() && value == null && mapping.getDefaultValue() == null) {
                    throw new RuntimeException("必填字段缺失: " + mapping.getSchoolField());
                }
                
            } catch (Exception e) {
                log.warn("字段映射失败: {} -> {}", 
                    mapping.getSchoolField(), mapping.getLocalField(), e);
            }
        }
        
        // 设置同步相关字段
        userSync.setSyncTime(LocalDateTime.now());
        userSync.setSyncSource(properties.getSource());
        userSync.setRawData(schoolUser.getRawData());
        
        return userSync;
    }
    
    /**
     * 获取字段值
     */
    private Object getFieldValue(Object obj, String fieldName) {
        try {
            // 支持嵌套属性访问，如 "college.name"
            String[] parts = fieldName.split("\\.");
            Object current = obj;
            
            for (String part : parts) {
                if (current == null) return null;
                
                Field field = current.getClass().getDeclaredField(part);
                field.setAccessible(true);
                current = field.get(current);
            }
            
            return current;
        } catch (Exception e) {
            return null;
        }
    }
    
    /**
     * 设置字段值
     */
    private void setFieldValue(Object obj, String fieldName, Object value) {
        try {
            Field field = obj.getClass().getDeclaredField(fieldName);
            field.setAccessible(true);
            field.set(obj, value);
        } catch (Exception e) {
            log.warn("设置字段值失败: {} = {}", fieldName, value, e);
        }
    }
    
    /**
     * 值转换
     */
    private Object convertValue(Object value, FieldMapping mapping) {
        if (mapping.getConverterType() == null) {
            return value;
        }
        
        switch (mapping.getConverterType()) {
            case "DATE":
                return convertToDate(value.toString());
            case "BOOLEAN":
                return convertToBoolean(value.toString());
            case "SPLIT":
                return splitField(value.toString(), mapping.getSeparator());
            case "JSON":
                return parseJson(value.toString());
            default:
                return value;
        }
    }
    
    private LocalDateTime convertToDate(String dateStr) {
        try {
            // 支持多种日期格式
            DateTimeFormatter[] formatters = {
                DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss"),
                DateTimeFormatter.ofPattern("yyyy/MM/dd HH:mm"),
                DateTimeFormatter.ofPattern("yyyyMMdd")
            };
            
            for (DateTimeFormatter formatter : formatters) {
                try {
                    return LocalDateTime.parse(dateStr, formatter);
                } catch (Exception ignored) {
                    // 继续尝试下一个格式
                }
            }
            
            return null;
        } catch (Exception e) {
            log.warn("日期转换失败: {}", dateStr);
            return null;
        }
    }
    
    private Boolean convertToBoolean(String value) {
        if (value == null) return false;
        
        String lowerValue = value.toLowerCase();
        return Arrays.asList("true", "1", "yes", "on", "是", "启用").contains(lowerValue);
    }
    
    private List<String> splitField(String value, String separator) {
        if (value == null || value.isEmpty()) return new ArrayList<>();
        
        String sep = separator != null ? separator : ",";
        return Arrays.asList(value.split(sep)).stream()
            .map(String::trim)
            .filter(s -> !s.isEmpty())
            .collect(Collectors.toList());
    }
    
    private Object parseJson(String jsonStr) {
        try {
            ObjectMapper mapper = new ObjectMapper();
            return mapper.readValue(jsonStr, Map.class);
        } catch (Exception e) {
            log.warn("JSON解析失败: {}", jsonStr);
            return null;
        }
    }
}
```

```java
@Component
@EnableScheduling
@ConditionalOnProperty(name = "school.data.sync.enabled", havingValue = "true")
public class DataSyncScheduler {
    
    @Autowired
    private DataSyncService dataSyncService;
    
    @Autowired
    private SchoolDataProperties properties;
    
    private static final Logger log = LoggerFactory.getLogger(DataSyncScheduler.class);
    
    /**
     * 定时同步任务
     */
    @Scheduled(cron = "${school.data.sync.cron}")
    public void scheduledSync() {
        log.info("开始执行定时数据同步");
        
        try {
            // 执行增量同步
            SyncResult result = dataSyncService.performIncrementalSync();
            
            log.info("增量同步完成: 新增{}, 更新{}, 错误{}", 
                result.getCreatedRecords(), 
                result.getUpdatedRecords(), 
                result.getErrorRecords());
            
            // 检查是否需要全量同步
            if (shouldPerformFullSync(result)) {
                log.info("触发全量同步");
                SyncResult fullResult = dataSyncService.performFullSync();
                log.info("全量同步完成: 新增{}, 更新{}, 错误{}", 
                    fullResult.getCreatedRecords(), 
                    fullResult.getUpdatedRecords(), 
                    fullResult.getErrorRecords());
            }
            
        } catch (Exception e) {
            log.error("定时同步失败", e);
        }
    }
    
    /**
     * 启动时执行同步（可选）
     */
    @EventListener(ApplicationReadyEvent.class)
    public void onApplicationReady() {
        if (properties.getSync().isFullSyncOnStartup()) {
            log.info("应用启动时执行全量同步");
            
            CompletableFuture.runAsync(() -> {
                try {
                    SyncResult result = dataSyncService.performFullSync();
                    log.info("启动全量同步完成: {}", result);
                } catch (Exception e) {
                    log.error("启动全量同步失败", e);
                }
            });
        }
    }
    
    /**
     * 判断是否需要全量同步
     */
    private boolean shouldPerformFullSync(SyncResult incrementalResult) {
        // 如果增量同步失败次数过多，触发全量同步
        if (incrementalResult.getErrorRecords() > incrementalResult.getTotalRecords() * 0.1) {
            return true;
        }
        
        // 如果增量同步记录数为0，可能是首次同步或数据异常
        if (incrementalResult.getTotalRecords() == 0) {
            return true;
        }
        
        return false;
    }
}
```

### 8.1 同步监控接口

#### SyncMonitorController.java
```java
@RestController
@RequestMapping("/api/admin/sync")
@PreAuthorize("hasAnyRole('ADMIN', 'OPS')")
public class SyncMonitorController {
    
    @Autowired
    private DataSyncService dataSyncService;
    
    @Autowired
    private SyncLogRepository syncLogRepository;
    
    /**
     * 获取同步状态
     */
    @GetMapping("/status")
    public ResponseEntity<?> getSyncStatus() {
        // 获取最近一次同步
        Optional<SyncLog> lastSync = syncLogRepository
            .findFirstByOrderByStartTimeDesc();
        
        Map<String, Object> status = new HashMap<>();
        status.put("lastSync", lastSync.map(this::convertToDto).orElse(null));
        status.put("isRunning", lastSync.map(log -> "running".equals(log.getStatus())).orElse(false));
        
        return ResponseEntity.ok(status);
    }
    
    /**
     * 手动触发同步
     */
    @PostMapping("/trigger")
    public ResponseEntity<?> triggerSync(@RequestParam(defaultValue = "incremental") String type) {
        try {
            SyncResult result;
            if ("full".equals(type)) {
                result = dataSyncService.performFullSync();
            } else {
                result = dataSyncService.performIncrementalSync();
            }
            
            return ResponseEntity.ok(Map.of(
                "message", "同步完成",
                "result", result
            ));
            
        } catch (Exception e) {
            return ResponseEntity.internalServerError().body(Map.of(
                "message", "同步失败: " + e.getMessage()
            ));
        }
    }
    
    /**
     * 获取同步历史
     */
    @GetMapping("/history")
    public ResponseEntity<?> getSyncHistory(
            @RequestParam(defaultValue = "0") int page,
            @RequestParam(defaultValue = "20") int size) {
        
        Page<SyncLog> syncLogs = syncLogRepository
            .findAllByOrderByStartTimeDesc(PageRequest.of(page, size));
        
        return ResponseEntity.ok(Map.of(
            "content", syncLogs.getContent().stream()
                .map(this::convertToDto).collect(Collectors.toList()),
            "total", syncLogs.getTotalElements(),
            "page", page,
            "size", size
        ));
    }
    
    private SyncLogDto convertToDto(SyncLog syncLog) {
        SyncLogDto dto = new SyncLogDto();
        dto.setId(syncLog.getId());
        dto.setSyncType(syncLog.getSyncType());
        dto.setSyncSource(syncLog.getSyncSource());
        dto.setStartTime(syncLog.getStartTime());
        dto.setEndTime(syncLog.getEndTime());
        dto.setTotalRecords(syncLog.getTotalRecords());
        dto.setUpdatedRecords(syncLog.getUpdatedRecords());
        dto.setCreatedRecords(syncLog.getCreatedRecords());
        dto.setErrorRecords(syncLog.getErrorRecords());
        dto.setStatus(syncLog.getStatus());
        dto.setErrorMessage(syncLog.getErrorMessage());
        return dto;
    }
}
```
