# 统一身份认证集成开发指南

> **目标**：实现基于内蒙古农业大学统一身份认证平台的单点登录  
> **协议支持**：CAS 2.0、OAuth 2.0  
> **开发周期**：5-7个工作日

#### CasSsoService实现
```java
@Service
@Profile("sso")
@ConditionalOnProperty(name = "sso.protocol", havingValue = "cas")
public class CasSsoService implements SsoService {
    
    @Autowired
    private SsoUserMappingService userMappingService;
    
    @Autowired
    private SsoProperties ssoProperties;
    
    @Override
    public AuthUrlResponse getLoginUrl(HttpSession session) {
        String loginUrl = ssoProperties.getCas().getServerUrl() + "/login" +
            "?service=" + URLEncoder.encode(ssoProperties.getCas().getServiceUrl(), StandardCharsets.UTF_8);
        
        return new AuthUrlResponse(loginUrl);
    }
    
    @Override
    public User authenticate(String ticket, String service, HttpSession session) {
        try {
            // 验证CAS ticket
            CasUser casUser = validateTicket(ticket, service);
            
            // 映射用户
            return userMappingService.findOrCreateUser(casUser, "CAS");
            
        } catch (Exception e) {
            throw new BusinessException("CAS认证失败: " + e.getMessage());
        }
    }
    
    private CasUser validateTicket(String ticket, String service) {
        String validateUrl = ssoProperties.getCas().getServerUrl() + "/validate";
        
        RestTemplate restTemplate = new RestTemplate();
        String response = restTemplate.getForObject(
            validateUrl + "?ticket=" + ticket + "&service=" + service,
            String.class
        );
        
        if (response != null && response.startsWith("yes\n")) {
            String username = response.split("\n")[1];
            
            // 获取用户属性
            Map<String, Object> attributes = getUserAttributes(username);
            
            return new CasUser(username, attributes);
        }
        
        throw new BusinessException("CAS ticket验证失败");
    }
    
    private Map<String, Object> getUserAttributes(String username) {
        // 从CAS获取用户属性
        String attributesUrl = ssoProperties.getCas().getServerUrl() + "/attributes";
        RestTemplate restTemplate = new RestTemplate();
        
        try {
            ResponseEntity<Map> response = restTemplate.getForEntity(
                attributesUrl + "?username=" + username,
                Map.class
            );
            
            return response.getBody();
        } catch (Exception e) {
            log.warn("获取用户属性失败: {}", e.getMessage());
            return new HashMap<>();
        }
    }
}
```

#### OAuth2SsoService实现
```java
@Service
@Profile("sso")
@ConditionalOnProperty(name = "sso.protocol", havingValue = "oauth2")
public class OAuth2SsoService implements SsoService {
    
    @Autowired
    private SsoUserMappingService userMappingService;
    
    @Autowired
    private SsoProperties ssoProperties;
    
    @Override
    public AuthUrlResponse getLoginUrl(HttpSession session) {
        String state = UUID.randomUUID().toString();
        String nonce = UUID.randomUUID().toString();
        
        session.setAttribute("sso_state", state);
        session.setAttribute("sso_nonce", nonce);
        
        String authUrl = String.format(
            "%s?response_type=code&client_id=%s&redirect_uri=%s&scope=%s&state=%s&nonce=%s",
            ssoProperties.getOauth2().getIssuerUri() + "/authorize",
            ssoProperties.getOauth2().getClientId(),
            URLEncoder.encode(ssoProperties.getOauth2().getRedirectUri(), StandardCharsets.UTF_8),
            String.join(" ", ssoProperties.getOauth2().getScopes()),
            state,
            nonce
        );
        
        return new AuthUrlResponse(authUrl);
    }
    
    @Override
    public User authenticate(String code, String state, HttpSession session) {
        // 验证state
        String savedState = (String) session.getAttribute("sso_state");
        if (!state.equals(savedState)) {
            throw new BusinessException("Invalid state");
        }
        
        // 换取token
        OidcIdToken idToken = exchangeCodeForToken(code);
        
        // 映射用户
        return userMappingService.findOrCreateUser(idToken, "OAuth2");
    }
    
    private OidcIdToken exchangeCodeForToken(String code) {
        RestTemplate restTemplate = new RestTemplate();
        
        Map<String, String> params = new HashMap<>();
        params.put("grant_type", "authorization_code");
        params.put("client_id", ssoProperties.getOauth2().getClientId());
        params.put("client_secret", ssoProperties.getOauth2().getClientSecret());
        params.put("code", code);
        params.put("redirect_uri", ssoProperties.getOauth2().getRedirectUri());
        
        HttpHeaders headers = new HttpHeaders();
        headers.setContentType(MediaType.APPLICATION_FORM_URLENCODED);
        
        HttpEntity<Map<String, String>> request = new HttpEntity<>(params, headers);
        
        ResponseEntity<Map> response = restTemplate.postForEntity(
            ssoProperties.getOauth2().getIssuerUri() + "/accessToken",
            request,
            Map.class
        );
        
        Map<String, Object> tokenResponse = response.getBody();
        String idTokenString = (String) tokenResponse.get("id_token");
        
        return OidcIdToken.withTokenValue(idTokenString).build();
    }
}
```

### 3.3 角色映射服务

#### RoleMappingService.java
```java
@Service
@Profile("sso")
public class RoleMappingService {
    
    @Autowired
    private SsoProperties ssoProperties;
    
    public String inferRoleFromMemberOf(List<String> memberOf) {
        if (memberOf == null || memberOf.isEmpty()) {
            return "STUDENT";
        }
        
        // 高权限角色优先匹配
        for (String org : memberOf) {
            if (isAdminOrganization(org)) {
                return "ADMIN";
            }
        }
        
        for (String org : memberOf) {
            if (isOpsOrganization(org)) {
                return "OPS";
            }
        }
        
        for (String org : memberOf) {
            if (isTeacherOrganization(org)) {
                return "TEACHER";
            }
        }
        
        return "STUDENT";
    }
    
    private boolean isAdminOrganization(String org) {
        String[] adminKeywords = {
            "管理", "行政", "办公室", "信息中心", "网络中心", 
            "教务处", "学生处", "人事处", "财务处"
        };
        
        return containsKeyword(org, adminKeywords);
    }
    
    private boolean isOpsOrganization(String org) {
        String[] opsKeywords = {
            "运维", "技术", "网络", "信息", "系统", "平台",
            "数据中心", "实验室", "设备"
        };
        
        return containsKeyword(org, opsKeywords);
    }
    
    private boolean isTeacherOrganization(String org) {
        String[] teacherKeywords = {
            "教师", "教学", "学院", "系部", "研究所", "教研室",
            "教授", "副教授", "讲师", "导师"
        };
        
        return containsKeyword(org, teacherKeywords);
    }
    
    private boolean containsKeyword(String org, String[] keywords) {
        for (String keyword : keywords) {
            if (org.contains(keyword)) {
                return true;
            }
        }
        return false;
    }
    
    public boolean requiresManualConfirmation(String role, List<String> memberOf) {
        if (ssoProperties.getRoleMapping().getRequiredForConfirmation().contains(role)) {
            return true;
        }
        
        if (memberOf != null) {
            for (String org : memberOf) {
                if (org.contains("临时") || org.contains("试用") || org.contains("实习")) {
                    return true;
                }
            }
        }
        
        return false;
    }
}
```

### 4.1 SsoAuthController.java

```java
@RestController
@RequestMapping("/api/auth/sso")
@Profile("sso")
public class SsoAuthController {
    
    @Autowired
    private SsoService ssoService;
    
    @Autowired
    private JwtUtil jwtUtil;
    
    @Autowired
    private RedisTemplate<String, String> redisTemplate;
    
    @GetMapping("/login")
    public ResponseEntity<?> login(HttpSession session) {
        if (!ssoService.isEnabled()) {
            return ResponseEntity.badRequest().body(Map.of("message", "SSO未启用"));
        }
        
        AuthUrlResponse response = ssoService.getLoginUrl(session);
        return ResponseEntity.ok(response);
    }
    
    @GetMapping("/callback")
    public void callback(
            @RequestParam(required = false) String code,
            @RequestParam(required = false) String state,
            @RequestParam(required = false) String ticket,
            @RequestParam(required = false) String service,
            HttpSession session,
            HttpServletResponse response) throws IOException {
        
        try {
            User user;
            
            if (code != null && state != null) {
                // OAuth 2.0 流程
                user = ssoService.authenticate(code, state, session);
            } else if (ticket != null && service != null) {
                // CAS 流程
                user = ssoService.authenticate(ticket, service, session);
            } else {
                throw new BusinessException("无效的回调参数");
            }
            
            // 生成JWT
            String token = jwtUtil.generateToken(user);
            
            // 生成一次性授权码
            String oneTimeCode = UUID.randomUUID().toString();
            redisTemplate.opsForValue().set(
                "sso:code:" + oneTimeCode, 
                token, 
                5, 
                TimeUnit.MINUTES
            );
            
            // 重定向到前端回调页
            String redirectUrl = "/sso/callback?code=" + oneTimeCode;
            response.sendRedirect(redirectUrl);
            
        } catch (Exception e) {
            log.error("SSO回调处理失败", e);
            response.sendRedirect("/login?error=sso_failed");
        }
    }
    
    @PostMapping("/token")
    public ResponseEntity<?> exchangeToken(@RequestBody Map<String, String> request) {
        String oneTimeCode = request.get("code");
        
        if (oneTimeCode == null) {
            return ResponseEntity.badRequest().body(Map.of("message", "缺少授权码"));
        }
        
        String token = redisTemplate.opsForValue().get("sso:code:" + oneTimeCode);
        if (token == null) {
            return ResponseEntity.badRequest().body(Map.of("message", "授权码无效或已过期"));
        }
        
        // 删除一次性授权码
        redisTemplate.delete("sso:code:" + oneTimeCode);
        
        return ResponseEntity.ok(Map.of("token", token));
    }
    
    @GetMapping("/status")
    public ResponseEntity<?> status() {
        return ResponseEntity.ok(Map.of(
            "enabled", ssoService.isEnabled(),
            "protocol", ssoService instanceof CasSsoService ? "cas" : "oauth2"
        ));
    }
    
    @PostMapping("/logout")
    public ResponseEntity<?> logout(@RequestBody Map<String, String> request, 
                                 HttpServletResponse response) {
        String token = request.get("token");
        
        try {
            ssoService.logout(token, response);
            return ResponseEntity.ok(Map.of("message", "退出成功"));
        } catch (Exception e) {
            log.error("SSO退出失败", e);
            return ResponseEntity.internalServerError().body(Map.of("message", "退出失败"));
        }
    }
}
```

## 六、配置文件

### 6.1 application-sso.yml

```yaml
sso:
  protocol: ${SSO_PROTOCOL:oauth2}  # oauth2 或 cas
  enabled: ${SSO_ENABLED:false}
  
  # OAuth 2.0 配置
  oauth2:
    client-id: ${SSO_CLIENT_ID}
    client-secret: ${SSO_CLIENT_SECRET}
    issuer-uri: https://ids.imau.edu.cn/authserver
    redirect-uri: ${APP_BASE_URL}/api/auth/sso/callback
    scopes: openid,profile,email
  
  # CAS 配置
  cas:
    server-url: https://ids.imau.edu.cn/authserver
    service-url: ${APP_BASE_URL}/api/auth/sso/callback
  
  # 角色映射配置
  role-mapping:
    auto-assign: true
    strict: false
    required-for-confirmation: ADMIN,OPS
    auto-approve: STUDENT,TEACHER
```

### 6.2 环境变量配置

```bash
# 生产环境
SSO_PROTOCOL=oauth2

SSO_CLIENT_SECRET=your_client_secret

# 开发环境（禁用SSO）
```
